# ── 수강료 청구서 자동 생성 스크립트 ────────────────── # Claude Code가 생성한 Python 스크립트 # Make.com HTTP 모듈에서 호출됩니다 import anthropic import json from flask import Flask, request, jsonify app = Flask(__name__) client = anthropic.Anthropic() @app.route('/generate-invoice', methods=['POST']) def generate_invoice(): data = request.get_json() # Make.com에서 전달받은 수강생 데이터 student_name = data['student_name'] grade = data['grade'] subject = data['subject'] amount = data['amount'] due_date = data['due_date'] parent_name = data['parent_name'] # Claude API 호출 → 청구서 문구 생성 message = client.messages.create( model="claude-haiku-4-5", max_tokens=1024, messages=[{ "role": "user", "content": f""" 학원 수강료 청구 이메일을 작성해주세요. 수신자: {parent_name}님 학생: {student_name} ({grade}) 수강과목: {subject} 청구금액: {amount}원 납부기한: {due_date} 형식: 정중하고 따뜻한 톤, 200자 이내 """ }] ) invoice_text = message.content[0].text return jsonify({ "success": True, "invoice_text": invoice_text, "email_subject": f"[수강료 안내] {student_name} {due_date[:7]}분" }) if __name__ == '__main__': app.run(port=5000)
Claude Code 프롬프트 · 위 스크립트를 생성하는 명령어
"학원 수강료 청구서를 자동 생성하는 Flask API를 만들어줘. Make.com에서 POST 요청으로 학생 정보를 받아서 Claude Haiku API로 청구서 이메일 문구를 생성하고 JSON으로 반환하는 /generate-invoice 엔드포인트가 필요해. 입력: student_name, grade, subject, amount, due_date, parent_name"
